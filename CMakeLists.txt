cmake_minimum_required(VERSION 3.22)
project(Phosrack VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network Widgets)
if(WIN32)
    add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    set(PHOSRACK_AUDIO_SOURCE src/audioanalyzer_windows.cpp)
    set(PHOSRACK_EQ_SOURCE src/equalizer_windows.cpp)
    set(PHOSRACK_EXECUTABLE_DESTINATION ".")
    if(MSVC)
        add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
    endif()
else()
    find_package(KF6Wallet 6.0 REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PipeWire REQUIRED IMPORTED_TARGET libpipewire-0.3)
    set(PHOSRACK_AUDIO_SOURCE src/audioanalyzer.cpp)
    set(PHOSRACK_EQ_SOURCE src/equalizer.cpp)
    set(PHOSRACK_EXECUTABLE_DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif()

add_library(phosrack-core STATIC
    src/protocol.cpp src/protocol.h src/spectrum.cpp src/spectrum.h
    src/spotifyclient.cpp src/spotifyclient.h ${PHOSRACK_AUDIO_SOURCE} src/audioanalyzer.h
    src/preferences.cpp src/preferences.h src/localization.cpp src/localization.h
    ${PHOSRACK_EQ_SOURCE} src/equalizer.h src/tokenstore.cpp src/tokenstore.h
    src/apoconfig.cpp src/apoconfig.h)
target_link_libraries(phosrack-core PUBLIC Qt6::Core Qt6::Gui Qt6::Network)
if(WIN32)
    target_link_libraries(phosrack-core PUBLIC advapi32 ole32 uuid)
else()
    target_link_libraries(phosrack-core PUBLIC KF6::Wallet)
endif()
target_include_directories(phosrack-core PUBLIC src)
if(MSVC)
    target_compile_options(phosrack-core PRIVATE /W4)
else()
    target_compile_options(phosrack-core PRIVATE -Wall -Wextra -Wpedantic)
endif()

qt_add_executable(phosrack src/main.cpp)
target_link_libraries(phosrack PRIVATE phosrack-core Qt6::Widgets Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(phosrack PRIVATE PHOSRACK_VERSION="${PROJECT_VERSION}")
if(WIN32)
    set_target_properties(phosrack PROPERTIES WIN32_EXECUTABLE TRUE)
else()
    add_executable(phosrack-eq-worker src/equalizer_worker.cpp)
    target_link_libraries(phosrack-eq-worker PRIVATE Qt6::Core PkgConfig::PipeWire)
    add_dependencies(phosrack phosrack-eq-worker)
    install(TARGETS phosrack-eq-worker RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
file(GLOB PHOSRACK_QML CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/package/contents/ui/*.qml")
qt_add_resources(phosrack "phosrack-ui"
    PREFIX "/ui" BASE "${CMAKE_SOURCE_DIR}/package/contents/ui" FILES ${PHOSRACK_QML})
file(GLOB PHOSRACK_TRANSLATIONS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/translations/*.json")
if(PHOSRACK_TRANSLATIONS)
    qt_add_resources(phosrack "phosrack-i18n"
        PREFIX "/i18n" BASE "${CMAKE_SOURCE_DIR}/translations" FILES ${PHOSRACK_TRANSLATIONS})
endif()
qt_add_resources(phosrack "phosrack-icon" PREFIX "/" BASE "${CMAKE_SOURCE_DIR}/app"
    FILES app/io.phosrack.Phosrack.svg)

install(TARGETS phosrack RUNTIME DESTINATION ${PHOSRACK_EXECUTABLE_DESTINATION})
if(NOT WIN32)
    install(FILES app/io.phosrack.Phosrack.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES app/io.phosrack.Phosrack.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
    install(FILES app/io.phosrack.Phosrack.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/phosrack)
install(FILES README.md GETTING_STARTED.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/phosrack)
set(PHOSRACK_WINDOWS_DEPLOY_DIR "" CACHE PATH "Staged Qt Windows deployment copied into installer packages")
if(WIN32 AND PHOSRACK_WINDOWS_DEPLOY_DIR)
    install(DIRECTORY "${PHOSRACK_WINDOWS_DEPLOY_DIR}/" DESTINATION .)
endif()

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test QuickTest)
    add_executable(phosrack-tests tests/core_test.cpp src/protocol.cpp src/spectrum.cpp)
    target_include_directories(phosrack-tests PRIVATE src)
    target_link_libraries(phosrack-tests PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME core COMMAND phosrack-tests)
    add_executable(phosrack-spotify-tests tests/spotify_test.cpp)
    target_link_libraries(phosrack-spotify-tests PRIVATE phosrack-core Qt6::Test)
    add_test(NAME spotify-protocol COMMAND phosrack-spotify-tests)
    set_tests_properties(spotify-protocol PROPERTIES ENVIRONMENT "PHOSRACK_TEST_MODE=1")
    if(EXISTS "${CMAKE_SOURCE_DIR}/tests/localization_test.cpp")
        add_executable(phosrack-localization-tests tests/localization_test.cpp)
        target_link_libraries(phosrack-localization-tests PRIVATE phosrack-core Qt6::Test)
        if(PHOSRACK_TRANSLATIONS)
            qt_add_resources(phosrack-localization-tests "phosrack-test-i18n"
                PREFIX "/i18n" BASE "${CMAKE_SOURCE_DIR}/translations" FILES ${PHOSRACK_TRANSLATIONS})
        endif()
        add_test(NAME localization COMMAND phosrack-localization-tests)
    endif()
    if(NOT WIN32 AND EXISTS "${CMAKE_SOURCE_DIR}/tests/audio_test.cpp")
        add_executable(phosrack-audio-tests tests/audio_test.cpp)
        target_link_libraries(phosrack-audio-tests PRIVATE phosrack-core Qt6::Test)
        add_test(NAME audio-routing COMMAND phosrack-audio-tests)
        set_tests_properties(audio-routing PROPERTIES ENVIRONMENT "PHOSRACK_TEST_MODE=1")
    endif()
    add_executable(phosrack-ui-tests tests/ui_test.cpp)
    target_link_libraries(phosrack-ui-tests PRIVATE Qt6::QuickTest)
    add_test(NAME ui-controls COMMAND phosrack-ui-tests -input "${CMAKE_SOURCE_DIR}/tests/qml")
    set_tests_properties(ui-controls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Basic" TIMEOUT 20)
    if(NOT WIN32 AND EXISTS "${CMAKE_SOURCE_DIR}/tests/equalizer_test.cpp")
        add_executable(phosrack-equalizer-tests tests/equalizer_test.cpp)
        target_link_libraries(phosrack-equalizer-tests PRIVATE phosrack-core Qt6::Test)
        add_test(NAME equalizer COMMAND phosrack-equalizer-tests)
        set_tests_properties(equalizer PROPERTIES ENVIRONMENT "PHOSRACK_TEST_MODE=1")
    endif()
    foreach(test_name IN ITEMS tokenstore monitorformat apo)
        if(EXISTS "${CMAKE_SOURCE_DIR}/tests/${test_name}_test.cpp")
            add_executable(phosrack-${test_name}-tests tests/${test_name}_test.cpp)
            target_link_libraries(phosrack-${test_name}-tests PRIVATE phosrack-core Qt6::Test)
            if(test_name STREQUAL "tokenstore")
                set(test_label "token-store")
            elseif(test_name STREQUAL "monitorformat")
                set(test_label "monitor-format")
            else()
                set(test_label "apo-config")
            endif()
            add_test(NAME ${test_label} COMMAND phosrack-${test_name}-tests)
            set_tests_properties(${test_label} PROPERTIES ENVIRONMENT "PHOSRACK_TEST_MODE=1")
        endif()
    endforeach()
    if(WIN32 AND EXISTS "${CMAKE_SOURCE_DIR}/tests/equalizer_windows_test.cpp")
        add_executable(phosrack-windows-equalizer-tests tests/equalizer_windows_test.cpp)
        target_link_libraries(phosrack-windows-equalizer-tests PRIVATE phosrack-core Qt6::Test)
        add_test(NAME windows-equalizer COMMAND phosrack-windows-equalizer-tests)
        set_tests_properties(windows-equalizer PROPERTIES ENVIRONMENT "PHOSRACK_TEST_MODE=1")
    endif()
    foreach(language IN ITEMS en es fr de)
        add_test(NAME qml-smoke-${language} COMMAND phosrack --smoke-test --settings --language ${language})
        set_tests_properties(qml-smoke-${language} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;PHOSRACK_TEST_MODE=1" TIMEOUT 20)
        add_test(NAME fullscreen-smoke-${language} COMMAND phosrack --smoke-test --visualizer --language ${language})
        set_tests_properties(fullscreen-smoke-${language} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;PHOSRACK_TEST_MODE=1" TIMEOUT 20)
    endforeach()
endif()

set(CPACK_PACKAGE_NAME "phosrack")
set(CPACK_PACKAGE_VENDOR "Phosrack contributors")
set(CPACK_PACKAGE_CONTACT "Phosrack contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "A compact modular Spotify controller and local audio visualizer")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_PACKAGE_FILE_NAME "phosrack-${PROJECT_VERSION}-${CMAKE_SYSTEM_PROCESSOR}")
if(WIN32)
    set(CPACK_PACKAGE_FILE_NAME "phosrack-${PROJECT_VERSION}-windows-x64")
    set(CPACK_GENERATOR "ZIP;NSIS")
    set(CPACK_PACKAGE_INSTALL_DIRECTORY "Phosrack")
    set(CPACK_NSIS_PACKAGE_NAME "Phosrack")
    set(CPACK_NSIS_DISPLAY_NAME "Phosrack")
    set(CPACK_NSIS_MODIFY_PATH OFF)
    set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
    set(CPACK_NSIS_EXECUTABLES_DIRECTORY ".")
    set(CPACK_PACKAGE_EXECUTABLES "phosrack" "Phosrack")
    set(CPACK_NSIS_MUI_FINISHPAGE_RUN "phosrack.exe")
endif()
set(CPACK_RPM_PACKAGE_LICENSE "GPL-3.0-or-later")
set(CPACK_RPM_PACKAGE_GROUP "Applications/Multimedia")
set(CPACK_RPM_PACKAGE_RELEASE "1")
set(CPACK_RPM_PACKAGE_RELEASE_DIST ON)
set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
set(CPACK_RPM_PACKAGE_REQUIRES "qt6-qtdeclarative, qt6-qtsvg, qt6-qtwayland, pipewire, pipewire-libs, pipewire-utils")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION
    "/usr/share/licenses" "/usr/share/metainfo" "/usr/share/icons"
    "/usr/share/icons/hicolor" "/usr/share/icons/hicolor/scalable"
    "/usr/share/icons/hicolor/scalable/apps")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SECTION "sound")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-templates, qml6-module-qtquick-layouts, qml6-module-qtquick-window, qml6-module-qtqml, qml6-module-qtqml-models, qml6-module-qtqml-workerscript, libqt6svg6, qt6-wayland, pipewire-bin, libpipewire-0.3-modules")
set(CPACK_SOURCE_GENERATOR "TGZ")
set(CPACK_SOURCE_PACKAGE_FILE_NAME "phosrack-${PROJECT_VERSION}-source")
set(CPACK_SOURCE_IGNORE_FILES "/[.]git/;/releases/;/build[^/]*/;[.]user$;[.]swp$;[.]plasmoid$;[.]zip$;[.]rpm$;[.]deb$;[.]flatpak$;CMakeUserPresets.json")
include(CPack)
